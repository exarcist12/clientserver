CREATE TABLE item (
    id SERIAL PRIMARY KEY,
    title VARCHAR(50),
    price NUMERIC
);

CREATE TABLE category (
    id SERIAL PRIMARY KEY,
    category VARCHAR(50)
);

CREATE TABLE item_category (
    item_id INTEGER REFERENCES item(id),
    category_id INTEGER REFERENCES category(id),
    PRIMARY KEY (item_id, category_id)
);
